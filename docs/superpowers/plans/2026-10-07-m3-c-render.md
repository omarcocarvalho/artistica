# M3-C: Render Lines Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Put composition lines in the page model as page-millimetre vector paths (C1), draw them as vector strokes in the PDF and parse them back for tests (C2), and draw the same paths in the canvas preview (C3). Preview = PDF by construction: both renderers replay `PageModel.lines` and neither computes geometry (M3-R6).

**Architecture:** `buildPageModels` already knows each tile's trim and whether the engine turned the placement. C1 adds `compositionPaths(lines, frameOf(trim, turned))` (A2 + A3 geometry, canonical order), maps every command onto the page with `frameToPage`, and batches the paths of one tile into one solid stroke and one dashed stroke (M3-R7). C2 emits, per tile, a clipped graphics-state block of PDF path operators after the images and before the crop marks; the inspector gains a small content-stream interpreter that returns every stroked path with its colour, width, dash, opacity and clip. C3 extends `PageCtx` and replays the same blocks on the sheet canvas with the same clip, order and stroke state, scaling millimetres to device pixels.

**Tech Stack:** TypeScript 6, pdf-lib (`@pdfme/pdf-lib` 6.2.2, low-level operators), Vitest 5 + fast-check. No new dependencies.

**Spec:** `docs/spec.md` §2.5 ("lines and marks drawn as vectors, so they stay sharp"), §2.7 (on top of the image, on every printed version), §4 (one page model, two renderers; `PageModel` describes line segments), §7 ("PDF tests … check … that the vector lines are present") **and** `docs/superpowers/plans/2026-10-07-m3-overview.md` (M3-R2–R9, R15, R19; Shared contracts → Render; Memory and Performance budgets).

**Prerequisite:** A2 and A3 merged (C1). C2 and C3 start when C1 merges and run in parallel.

```ts
import type { LineSettings } from '../../shared/model/lines'
import { activeLineTypes } from '../../shared/model/lines'
import { armaturePaths, centrePaths, goldenPaths, gridPaths, thirdsPaths } from './geometry'
import { goldenSpiral } from './spiral'
import type { FramePath, FrameSize, PathCmd } from './types'

/** Every active type's paths in canonical order (M3-R15). Centre lines are the only dashed type (M3-R14). */
export function compositionPaths(lines: LineSettings, frame: FrameSize): FramePath[] {
  const cmdsOf = (type: FramePath['type']): readonly PathCmd[] => {
    switch (type) {
      case 'grid':
        return gridPaths(lines.grid.cols, lines.grid.rows, frame)
      case 'thirds':
        return thirdsPaths(frame)
      case 'armature':
        return armaturePaths(frame)
      case 'golden':
        return goldenPaths(frame)
      case 'spiral':
        return goldenSpiral(lines.spiral.corner, frame)
      case 'centre':
        return centrePaths(frame)
    }
  }
  return activeLineTypes(lines).flatMap((type) => {
    const cmds = cmdsOf(type)
    return cmds.length === 0 ? [] : [{ type, dashed: type === 'centre', cmds }]
  })
}
```

- [ ] **Step 5: Implement** `tile-lines.ts`:

```ts
import { centreDashMm, compositionPaths, frameOf, frameToPage } from '../../lines'
import { activeLineTypes, type LineSettings } from '../../../shared/model/lines'
import type { RectMm } from '../../layout/types'
import type { LineStroke, TileLines } from '../types'

export const MAX_LINE_CMDS_PER_TILE = 200

export function tileLinesFor(
  lines: LineSettings,
  trim: RectMm,
  turned: boolean,
  tileIndex: number,
): TileLines | null {
  const paths = compositionPaths(lines, frameOf(trim, turned))
  if (paths.length === 0) return null
  const place = (dashed: boolean) =>
    paths.filter((p) => p.dashed === dashed).flatMap((p) => p.cmds.map((c) => frameToPage(c, trim, turned)))
  const { colour, widthMm, opacityPct } = lines.style
  const strokes: LineStroke[] = [
    { dashMm: [], cmds: place(false) },
    { dashMm: [...centreDashMm(widthMm)], cmds: place(true) },
  ].filter((s) => s.cmds.length > 0)
  return { tileIndex, clip: trim, colour, opacity: opacityPct / 100, widthMm, types: activeLineTypes(lines), strokes }
}
```

```ts
interface GState {
  ctm: [number, number, number, number, number, number]
  colour: PdfStroke['colour']
  widthPt: number
  dashPt: number[]
  opacity: number
  clip: PdfStroke['clip']
  cap: number
  join: number
}

/** Tokenise a content stream into operands + operator (numbers, names, arrays; strings and inline images are skipped). */
function* operations(content: string): Generator<{ op: string; args: (number | string | number[])[] }> { /* … */ }

export function strokesOf(content: string, extGStateOpacity: (name: string) => number): PdfStroke[] {
  const stack: GState[] = []
  let gs: GState = { ctm: [1, 0, 0, 1, 0, 0], colour: { space: 'gray', values: [0] }, widthPt: 1, dashPt: [], opacity: 1, clip: null, cap: 0, join: 0 }
  let path: PdfPathOp[] = []
  let pendingClip: PdfStroke['clip'] = null
  const out: PdfStroke[] = []
  for (const { op, args } of operations(content)) {
    switch (op) {
      case 'q': stack.push(structuredClone(gs)); break
      case 'Q': gs = stack.pop() ?? gs; break
      case 'cm': gs.ctm = multiply(args as number[], gs.ctm); break
      case 'w': gs.widthPt = args[0] as number; break
      case 'J': gs.cap = args[0] as number; break
      case 'j': gs.join = args[0] as number; break
      case 'd': gs.dashPt = args[0] as number[]; break
      case 'RG': gs.colour = { space: 'rgb', values: args as number[] }; break
      case 'K': gs.colour = { space: 'cmyk', values: args as number[] }; break
      case 'G': gs.colour = { space: 'gray', values: args as number[] }; break
      case 'gs': gs.opacity = extGStateOpacity(args[0] as string); break
      case 're': { const [x, y, w, h] = args as number[]; pendingClip = { x, y, w, h }; path = []; break }
      case 'W': gs.clip = pendingClip; break
      case 'n': path = []; break
      case 'm': case 'l': { const [x, y] = apply(gs.ctm, args as number[]); path.push({ op, x, y }); break }
      case 'c': { /* apply ctm to the three points */ break }
      case 'S': out.push({ ...gs, colour: gs.colour, dashPt: [...gs.dashPt], path }); path = []; break
      default: break // rg/k/g (fill colours), Do, BT…ET: ignored
    }
  }
  return out
}
```

- [ ] **Step 5: Implement** in `compose.ts`:

```ts
import {
  appendBezierCurve, clip, endPath, lineTo, LineCapStyle, LineJoinStyle, moveTo,
  popGraphicsState, pushGraphicsState, rectangle, setDashPattern, setGraphicsState,
  setLineCap, setLineJoin, setLineWidth, setStrokingRgbColor, stroke, PDFName, type PDFPage,
} from '@pdfme/pdf-lib'

const hexToRgb = (hex: string): [number, number, number] =>
  [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255) as [number, number, number]

/** M3-R6/R8: replay one TileLines block. pt, y flipped. */
function drawTileLines(doc: PDFDocument, page: PDFPage, tl: TileLines, H: number, gsFor: (opacity: number) => PDFName | null): void {
  const X = (x: number) => pt(x)
  const Y = (y: number) => pt(H - y)
  const r = toPdfRect(tl.clip, H)
  const [red, green, blue] = hexToRgb(tl.colour)
  const ops = [
    pushGraphicsState(),
    rectangle(r.x, r.y, r.width, r.height), clip(), endPath(),
    setStrokingRgbColor(red, green, blue),
    setLineWidth(pt(tl.widthMm)), setLineCap(LineCapStyle.Butt), setLineJoin(LineJoinStyle.Miter),
  ]
  const gs = gsFor(tl.opacity)
  if (gs) ops.push(setGraphicsState(gs))
  for (const s of tl.strokes) {
    ops.push(setDashPattern(s.dashMm.map(pt), 0))
    for (const c of s.cmds)
      ops.push(
        c.op === 'M' ? moveTo(X(c.x), Y(c.y))
        : c.op === 'L' ? lineTo(X(c.x), Y(c.y))
        : appendBezierCurve(X(c.x1), Y(c.y1), X(c.x2), Y(c.y2), X(c.x), Y(c.y)),
      )
    ops.push(stroke())
  }
  ops.push(popGraphicsState())
  page.pushOperators(...ops)
}
```

  }
